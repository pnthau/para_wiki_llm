 @Entity
    public class PeerRating {
        @Id
        @GeneratedValue(strategy = GenerationType.IDENTITY)
        private Long id;

        // JID (Rater ID): ID của người chấm điểm
        private Long raterId;

        // PID (Presenter ID): ID của người bị chấm điểm
        private Long rateeId;

        // Ratings (1-35): Tổng điểm từ 7 tiêu chí (mỗi tiêu chí 1-5 sao)
        private Integer totalScore;

        // J_Gen & P_Gen: Giới tính của 2 người (để phát hiện Gender Bias)
        private String raterGender;
        private String rateeGender;

        // Friend (1-5): Có quen biết từ trước không? (Rất quan trọng)
        // Trong app của bạn, nếu Matchmaking 2 người từng chat với nhau, trường này = 5.
        private Integer previousEncounters;

        // J_PresScores: Trình độ THẬT của người chấm (Do giáo viên chấm khi vượt cấp)
        // Dùng để đánh giá xem người chấm có đủ "trình" để điểm của họ được tính trọng
  số cao không.
        private Integer raterActualSkillLevel;
    }
